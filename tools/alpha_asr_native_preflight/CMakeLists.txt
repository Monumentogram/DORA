cmake_minimum_required(VERSION 3.22)
project(dora_asr_preflight LANGUAGES C CXX)
if(NOT ANDROID_ABI STREQUAL "arm64-v8a" OR NOT ANDROID_PLATFORM STREQUAL "android-28")
    message(FATAL_ERROR "Stage-0 preflight requires Android arm64 API28")
endif()
if(NOT DEFINED WHISPER_SOURCE OR NOT DEFINED CANDIDATE_LIB_DIR)
    message(FATAL_ERROR "Explicit pinned source and verified candidate directory required")
endif()
add_library(whisper SHARED IMPORTED)
set_target_properties(whisper PROPERTIES IMPORTED_LOCATION "${CANDIDATE_LIB_DIR}/libwhisper.so")
add_executable(dora-asr-preflight main.cpp)
target_include_directories(dora-asr-preflight PRIVATE "${WHISPER_SOURCE}/include" "${WHISPER_SOURCE}/ggml/include")
target_compile_features(dora-asr-preflight PRIVATE cxx_std_17)
target_compile_options(dora-asr-preflight PRIVATE -march=armv8-a -Wall -Wextra -Werror)
target_link_options(dora-asr-preflight PRIVATE "-Wl,-rpath-link,${CANDIDATE_LIB_DIR}")
target_link_libraries(dora-asr-preflight PRIVATE whisper dl)
