cmake_minimum_required(VERSION 3.22)
project(dora_asr_campaign LANGUAGES C CXX)
if(NOT ANDROID_ABI STREQUAL "arm64-v8a" OR NOT ANDROID_PLATFORM STREQUAL "android-28")
    message(FATAL_ERROR "Campaign requires Android arm64 API28")
endif()
if(NOT DEFINED WHISPER_SOURCE OR NOT DEFINED CANDIDATE_LIB_DIR)
    message(FATAL_ERROR "Explicit pinned source and candidate required")
endif()
add_library(whisper SHARED IMPORTED)
set_target_properties(whisper PROPERTIES IMPORTED_LOCATION "${CANDIDATE_LIB_DIR}/libwhisper.so")
add_library(campaign_decode STATIC decode.c)
target_include_directories(campaign_decode PRIVATE "${WHISPER_SOURCE}/examples")
target_compile_definitions(campaign_decode PRIVATE MA_NO_DEVICE_IO MA_NO_ENGINE MA_NO_RESOURCE_MANAGER MA_NO_NODE_GRAPH MA_NO_GENERATION MA_NO_ENCODING)
add_executable(dora-asr-campaign main.cpp)
target_include_directories(dora-asr-campaign PRIVATE "${WHISPER_SOURCE}/include" "${WHISPER_SOURCE}/ggml/include" "${WHISPER_SOURCE}/examples")
target_compile_features(dora-asr-campaign PRIVATE cxx_std_17)
target_compile_options(dora-asr-campaign PRIVATE -march=armv8-a -Wall -Wextra -Werror)
target_link_options(dora-asr-campaign PRIVATE "-Wl,-rpath-link,${CANDIDATE_LIB_DIR}" "-Wl,-z,max-page-size=16384")
target_link_libraries(dora-asr-campaign PRIVATE whisper campaign_decode dl m)
