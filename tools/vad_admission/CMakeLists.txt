cmake_minimum_required(VERSION 3.22.1)
project(dora_vad_admission LANGUAGES C CXX)
# Admission-only external build. The pinned upstream checkout is never patched.
if(NOT ANDROID_ABI STREQUAL "arm64-v8a" OR NOT ANDROID_PLATFORM STREQUAL "android-28")
  message(FATAL_ERROR "Only arm64-v8a / API28 is admitted by this profile")
endif()
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
file(STRINGS "${SHERPA_SOURCE}/CMakeLists.txt" option_lines REGEX "^option\\(SHERPA_ONNX_")
foreach(line IN LISTS option_lines)
  string(REGEX MATCH "option\\((SHERPA_ONNX_[A-Z0-9_]+)" match "${line}")
  set(${CMAKE_MATCH_1} OFF CACHE BOOL "" FORCE)
endforeach()
# Upstream forces full JNI on Android if both JNI and C_API are OFF.
# Keep the C_API target configured, but never build/package it. Build only our target.
set(SHERPA_ONNX_ENABLE_C_API ON CACHE BOOL "" FORCE)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -g0 -ffunction-sections -fdata-sections -ffile-prefix-map=${SHERPA_SOURCE}=upstream -ffile-prefix-map=${CMAKE_BINARY_DIR}=build")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -g0 -ffunction-sections -fdata-sections -ffile-prefix-map=${SHERPA_SOURCE}=upstream -ffile-prefix-map=${CMAKE_BINARY_DIR}=build")
add_subdirectory("${SHERPA_SOURCE}" upstream EXCLUDE_FROM_ALL)
if(SHERPA_ONNX_ENABLE_TTS OR SHERPA_ONNX_ENABLE_SPEAKER_DIARIZATION OR SHERPA_ONNX_ENABLE_JNI)
  message(FATAL_ERROR "Unexpected feature enabled")
endif()
add_library(dora-vad-jni SHARED "${SHERPA_SOURCE}/sherpa-onnx/jni/voice-activity-detector.cc" "${SHERPA_SOURCE}/sherpa-onnx/jni/version.cc")
target_include_directories(dora-vad-jni PRIVATE "${SHERPA_SOURCE}")
target_compile_definitions(dora-vad-jni PRIVATE SHERPA_ONNX_ENABLE_TTS=0 SHERPA_ONNX_ENABLE_SPEAKER_DIARIZATION=0)
target_link_libraries(dora-vad-jni PRIVATE sherpa-onnx-core)
target_link_options(dora-vad-jni PRIVATE "-Wl,--gc-sections" "-Wl,-z,max-page-size=16384" "-Wl,-Map,${CMAKE_BINARY_DIR}/vad-link.map" "-Wl,--version-script=${SHERPA_SOURCE}/sherpa-onnx/jni/sherpa-onnx-symbols.lds")
set_target_properties(dora-vad-jni PROPERTIES OUTPUT_NAME sherpa-onnx-jni CXX_VISIBILITY_PRESET hidden LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/candidate")
