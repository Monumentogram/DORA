cmake_minimum_required(VERSION 3.22)
project(dora_stage9_whisper LANGUAGES CXX)
option(DORA_TEST_X86_RUNTIME "Unadmitted disposable-emulator functional runtime only" OFF)
if(NOT ANDROID_PLATFORM STREQUAL "android-28")
    message(FATAL_ERROR "Stage9 bounded runtime requires Android arm64 API28")
endif()
if(DORA_TEST_X86_RUNTIME)
    if(NOT ANDROID_ABI STREQUAL "x86_64")
        message(FATAL_ERROR "The separate test adapter only supports x86_64")
    endif()
elseif(NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Product adapter requires arm64-v8a")
endif()
if(NOT DEFINED WHISPER_SOURCE)
    message(FATAL_ERROR "Explicit pinned headers required")
endif()
add_library(dora-whisper-jni SHARED whisper_jni.cpp)
target_include_directories(dora-whisper-jni PRIVATE "${WHISPER_SOURCE}/include" "${WHISPER_SOURCE}/ggml/include")
target_compile_features(dora-whisper-jni PRIVATE cxx_std_17)
target_compile_options(dora-whisper-jni PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
if(DORA_TEST_X86_RUNTIME)
    target_compile_definitions(dora-whisper-jni PRIVATE DORA_TEST_X86_RUNTIME=1)
    target_compile_options(dora-whisper-jni PRIVATE -march=x86-64)
else()
    target_compile_options(dora-whisper-jni PRIVATE -march=armv8-a)
endif()
target_link_options(dora-whisper-jni PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
# Deliberately no libwhisper DT_NEEDED: its ARM82 dependencies load only after the HWCAP guard.
target_link_libraries(dora-whisper-jni PRIVATE dl)
